CREATE TABLE circuits (
	circuitId INT IDENTITY(1,1) PRIMARY KEY,
	circuitRef NVARCHAR(50) NOT NULL,
	name NVARCHAR(50) NOT NULL,
	location NVARCHAR(50) NOT NULL,
	country NVARCHAR(50) NOT NULL,
	lat FLOAT NOT NULL,
	lng FLOAT NOT NULL,
	alt SMALLINT NOT NULL,
	url NVARCHAR(MAX) NOT NULL,
	createdAt DATETIME2(3) NOT NULL
		CONSTRAINT DF_circuits_createdAt DEFAULT SYSUTCDATETIME(),
	updatedAt DATETIME2(3) NOT NULL
		CONSTRAINT DF_circuits_updatedAt DEFAULT SYSUTCDATETIME()
);

CREATE TRIGGER trg_circuits_updatedAt
ON circuits
AFTER UPDATE
AS BEGIN
	SET NOCOUNT ON;
	
	UPDATE c
	SET updatedAt = SYSUTCDATETIME()
	FROM circuits c
	INNER JOIN inserted i
		ON c.circuitId = i.circuitId;
END;

SELECT
	c.name AS column_name,
	t.name AS data_type,
	c.max_length,
	c.is_nullable
FROM sys.columns c
JOIN sys.types t
	ON c.user_type_id = t.user_type_id
WHERE c.object_id = OBJECT_ID('circuits');

CREATE PROCEDURE CreateCircuitsTable
AS
BEGIN
	DROP TABLE IF EXISTS circuits
	
	CREATE TABLE circuits (
		circuitId INT IDENTITY(1,1) PRIMARY KEY,
		circuitRef NVARCHAR(50) NOT NULL,
		name NVARCHAR(50) NOT NULL,
		location NVARCHAR(50) NOT NULL,
		country NVARCHAR(50) NOT NULL,
		lat FLOAT NOT NULL,
		lng FLOAT NOT NULL,
		alt SMALLINT NOT NULL,
		url NVARCHAR(MAX) NOT NULL,
		createdAt DATETIME2(3) NOT NULL
			CONSTRAINT DF_circuits_createdAt DEFAULT SYSUTCDATETIME(),
		updatedAt DATETIME2(3) NOT NULL
			CONSTRAINT DF_circuits_updatedAt DEFAULT SYSUTCDATETIME()
	);
END;
